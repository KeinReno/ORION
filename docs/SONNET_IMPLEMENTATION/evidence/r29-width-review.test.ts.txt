import {it,expect} from "vitest";
import ExcelJS from "exceljs";
import {addChart,workbookStateToXlsx,type WorkbookState} from "./xlsx-codec";
it("R29 chart writer preserves 1-based column widths",async()=>{
 const source:WorkbookState={sheets:[{name:"Sales",rows:[[{value:"Region"},{value:"Total"}],[{value:"North"},{value:12.3}]],columnWidths:{1:20,2:25}}]};
 const state=addChart(source,{sheetId:"Sales",dataRange:"A1:B2",chartType:"bar"});
 const wb=new ExcelJS.Workbook();
 await wb.xlsx.load(await workbookStateToXlsx(state) as never);
 expect([wb.worksheets[0].getColumn(1).width,wb.worksheets[0].getColumn(2).width]).toEqual([20,25]);
});
