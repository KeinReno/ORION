import { open, login, openPanel } from "./lib.mjs";
const { browser, page, errors } = await open();
await login(page);
await openPanel(page, "Документы");
await page.getByText(/^Новая таблица/).first().click();
await page.waitForTimeout(5000);
const rows = page.locator("table tbody tr");
const put = async (r, c, v) => { const td = rows.nth(r).locator("td").nth(c + 1); await td.dblclick(); await page.keyboard.press("Control+A"); await page.keyboard.type(v); await page.keyboard.press("Tab"); await page.waitForTimeout(150); };
await put(0, 0, "Месяц"); await put(0, 1, "Сумма");
await put(1, 0, "Янв"); await put(1, 1, "10");
await put(2, 0, "Фев"); await put(2, 1, "20"); await put(3, 0, "Мар"); await put(3, 1, "30");
await page.getByRole("button", { name: "Вставить диаграмму" }).click();
await page.waitForTimeout(800);
await page.locator("input").filter({ hasText: "" }).nth(0);
await page.getByRole("button", { name: "Добавить" }).click();
await page.waitForTimeout(1500);
const saveBtn = page.getByRole("button", { name: "Сохранить", exact: true }); console.log("save buttons:", await saveBtn.count(), "enabled:", await saveBtn.first().isEnabled()); await saveBtn.first().click(); await page.waitForTimeout(1500); console.log("toasts:", await page.locator("[role=status], [role=alert], li").filter({ hasText: /Сохранено|Не удалось/ }).allTextContents());
await page.waitForTimeout(8000);
await page.screenshot({ path: "102-csp-chart-saved.png" }); console.log("status:", await page.getByText(/сохранен/i).allTextContents());
console.log("--- CSP/wasm violations:", errors.filter((e) => /Refused|Content Security|WebAssembly|wasm/i.test(e)).join("\n") || "none");
console.log("--- other errors:", errors.filter((e) => !/Refused|Content Security|WebAssembly|wasm/i.test(e)).slice(0, 10).join("\n") || "none");
await browser.close();
