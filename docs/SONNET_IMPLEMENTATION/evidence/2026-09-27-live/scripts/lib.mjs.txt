import { chromium } from "playwright";
export const BASE = "http://localhost:3010";
export async function open(opts = {}) {
  const browser = await chromium.launch();
  const context = await browser.newContext({
    viewport: opts.viewport ?? { width: 1440, height: 900 },
    locale: "ru-RU",
    timezoneId: opts.tz ?? "Europe/Moscow",
    reducedMotion: opts.reducedMotion ?? "no-preference",
    deviceScaleFactor: opts.scale ?? 1
  });
  const page = await context.newPage();
  const errors = [];
  page.on("pageerror", (e) => errors.push("pageerror: " + e.message));
  page.on("console", (m) => { if (m.type() === "error") errors.push("console: " + m.text().slice(0, 300)); });
  return { browser, context, page, errors };
}
export async function login(page, name = "Мария Волкова", password = "1111", space = "Демонстрационный 1") {
  await page.goto(`${BASE}/login`);
  await page.getByText(name, { exact: true }).click();
  const pw = page.getByPlaceholder("Пароль").first();
  await pw.waitFor({ timeout: 15000 });
  await pw.fill(password);
  await page.locator('button[type="submit"]').first().click();
  await page.waitForURL((u) => !u.pathname.startsWith("/login"), { timeout: 60000 });
  if (page.url().includes("/spaces")) {
    await page.getByText(space, { exact: true }).click();
    await page.waitForURL((u) => !u.pathname.startsWith("/spaces"), { timeout: 60000 });
  }
  await page.waitForTimeout(4000);
}
export async function openPanel(page, label) {
  await page.getByText(label, { exact: true }).first().click({ timeout: 8000 });
  await page.waitForTimeout(2500);
}
