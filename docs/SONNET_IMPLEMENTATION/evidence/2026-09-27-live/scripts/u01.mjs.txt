import { open, login, openPanel } from "./lib.mjs";
const { browser, page, errors } = await open();
await login(page);
const title = `U01 плёнка ${Date.now() % 100000}`;
// 1. capture from «Сегодня»
await page.getByPlaceholder(/В инбокс/).fill(title);
await page.keyboard.press("Enter");
await page.waitForTimeout(2000);
await page.screenshot({ path: "70-u01-captured.png" });
// 2. process into a task with a due date, stay in the queue
await openPanel(page, "Инбокс");
const card = page.locator("li, article, div").filter({ hasText: title }).last();
await page.getByText(title).first().waitFor({ timeout: 10000 });
await page.screenshot({ path: "71-u01-inbox.png" });
await page.getByRole("button", { name: "В задачу" }).first().click();
await page.waitForTimeout(800);
const due = page.locator('input[type="datetime-local"]').last();
await due.fill("2026-09-28T18:00");
await page.screenshot({ path: "72-u01-process.png" });
await page.getByRole("button", { name: "Создать" }).last().click();
await page.waitForTimeout(2500);
console.log("still in inbox:", page.url(), "remaining title visible:", await page.getByText(title).count());
await page.screenshot({ path: "73-u01-after-process.png" });
// 3. find the task on Dashboard / Tasks and start a timer
await openPanel(page, "Задачи");
const reset = page.getByRole("button", { name: /Сбросить|Все задачи/ });
await page.screenshot({ path: "74-u01-tasks.png" });
console.log("task visible in Tasks:", await page.getByText(title).count());
console.log(errors.join("\n"));
await browser.close();
